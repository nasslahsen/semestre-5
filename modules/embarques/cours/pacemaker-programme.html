Le programme complet et les livrables

Les questions 32 à 38 demandent de réaliser la maquette : acquérir l'ECG simulé, détecter, calculer la fréquence, cadencer par timer, stimuler avec une LED, envoyer les données au PC, surveiller par watchdog. Voici le programme qui réunit les briques des parties précédentes, prêt à téléverser sur une seule carte ESP32, puis ce que le rapport doit contenir.

L'ECG de test

Pour tester sans générateur, l'ESP32 fabrique elle-même un ECG simplifié sur son DAC (8 bits, 0 à 255 pour 0 à 3,3 V), à la fréquence frequenceTest :

  • une ligne de base vers 1 V (code 77) ;
  • une onde R pointue de 20 ms qui monte à 2,55 V (code 197) ;
  • une onde T arrondie, 250 ms après, qui monte à 1,5 V (code 117).

Avec T0, le cœur simulé s'arrête : la maquette doit alors stimuler à la fréquence cible. Avec T45 et une fréquence cible de 60 bpm, le cœur bat trop lentement : la maquette stimule entre ses battements propres.

Le programme

// Pacemaker simulé, mode VVI : stimule si aucun battement n'est détecté pendant l'intervalle d'échappement.
#include <esp_task_wdt.h>

// ---------- broches ----------
const int PIN_DAC  = 25;   // ECG de test (DAC1), relié à GPIO 34
const int PIN_ECG  = 34;   // ECG simulé (ADC1_CH6)
const int PIN_STIM = 2;    // LED de stimulation
const int PIN_DET  = 4;    // LED battement détecté
const int PIN_BAT  = 35;   // batterie, pont diviseur par 2 (ADC1_CH7)

// ---------- paramètres réglables (interface série) ----------
int frequenceCible = 60;   // bpm : intervalle d'échappement 1000 ms
int seuil = 2500;          // code ADC de détection de l'onde R
int refractaire = 250;     // ms
int frequenceTest = 70;    // bpm de l'ECG de test ; 0 = cœur arrêté
const int HYSTERESIS = 200;
const unsigned long INTERVALLE_MIN = 333;   // garde-fou : 180 bpm au plus
const float SEUIL_BAT = 3.5;                // V

// ---------- timers (question 35) ----------
hw_timer_t *timerEch = NULL;    // 2 ms : 500 échantillons par seconde
hw_timer_t *timerStim = NULL;   // intervalle d'échappement
volatile bool tickEch = false;
volatile bool tickStim = false;
void IRAM_ATTR onEch()  { tickEch = true; }
void IRAM_ATTR onStim() { tickStim = true; }

// ---------- état ----------
unsigned long dernierEvt = 0;   // ms : dernier battement, détecté ou stimulé
unsigned long dernierR = 0;     // ms : dernier battement détecté
unsigned long rr = 0;           // ms
float bpm = 0;
bool auDessus = false;
unsigned long finStim = 0, finDet = 0, dernierBilan = 0;
int nbStim = 0, stimulationsDeSuite = 0;
int vMin = 4095, vMax = 0;
float vBat = 0;
unsigned long phase = 0;        // ms dans le battement de l'ECG de test

// ---------- ECG de test sur le DAC ----------
void genererECG() {
  int code = 77;                                   // ligne de base, environ 1 V
  if (frequenceTest > 0) {
    unsigned long periode = 60000UL / frequenceTest;
    phase += 2;
    if (phase >= periode) phase = 0;
    if (phase < 10) code = 77 + phase * 12;               // montée de l'onde R
    else if (phase < 20) code = 197 - (phase - 10) * 12;  // descente
    else if (phase >= 250 && phase < 350) {               // onde T
      int d = abs((int)phase - 300);
      code = 117 - d * 40 / 50;
    }
  }
  dacWrite(PIN_DAC, code);
}

// ---------- réglage de la fréquence cible ----------
void reglerFrequence(int f) {
  frequenceCible = f;
  timerAlarm(timerStim, 60000000UL / f, true, 0);   // en µs
  timerWrite(timerStim, 0);
}

// ---------- détection (questions 33 et 34) ----------
bool detecter(int v, unsigned long t) {
  if (t - dernierEvt < (unsigned long)refractaire) return false;
  if (!auDessus && v > seuil) { auDessus = true; return true; }
  if (auDessus && v < seuil - HYSTERESIS) auDessus = false;
  return false;
}

void battementDetecte(unsigned long t) {
  if (dernierR != 0) {
    rr = t - dernierR;
    bpm = 60000.0 / rr;                            // question 34
  }
  dernierR = t;
  dernierEvt = t;
  stimulationsDeSuite = 0;
  timerWrite(timerStim, 0);                        // stimulation inhibée
  tickStim = false;
  digitalWrite(PIN_DET, HIGH);
  finDet = t + 20;
  Serial.printf("R,%lu,%lu,%.1f\n", t, rr, bpm);   // question 37
}

// ---------- stimulation (question 36) ----------
void stimuler(unsigned long t) {
  if (t - dernierEvt < INTERVALLE_MIN) return;    // garde-fou
  digitalWrite(PIN_STIM, HIGH);
  finStim = t + 20;                                // LED allumée 20 ms
  dernierEvt = t;
  nbStim++;
  stimulationsDeSuite++;
  Serial.printf("S,%lu\n", t);
}

// ---------- surveillance du signal ----------
void surveillerECG(int v, unsigned long t) {
  vMin = min(vMin, v);
  vMax = max(vMax, v);
  if (t - dernierBilan < 2000) return;
  if (vMax - vMin < 100) Serial.println("ALERTE,signal plat");
  if (vMin <= 0 || vMax >= 4095) Serial.println("ALERTE,saturation");
  if (stimulationsDeSuite > 5) Serial.println("ALERTE,pas de rythme propre");
  vBat = 2.0 * analogRead(PIN_BAT) * 3.3 / 4095.0;
  if (vBat < SEUIL_BAT) Serial.printf("ALERTE,batterie,%.2f\n", vBat);
  Serial.printf("ETAT,%d,%.1f,%d,%.2f\n", frequenceCible, bpm, nbStim, vBat);
  vMin = 4095; vMax = 0;
  dernierBilan = t;
}

// ---------- commandes série (question 21) ----------
void lireSerie() {
  if (!Serial.available()) return;
  String cmd = Serial.readStringUntil('\n');
  cmd.trim();
  if (cmd.length() == 0) return;
  char c = cmd.charAt(0);
  long val = cmd.substring(1).toInt();
  if (c == 'F' && val >= 30 && val <= 180) reglerFrequence(val);
  else if (c == 'S' && val >= 100 && val <= 4000) seuil = val;
  else if (c == 'R' && val >= 100 && val <= 500) refractaire = val;
  else if (c == 'T' && val >= 0 && val <= 200) frequenceTest = val;
  else if (c != '?') { Serial.println("ERREUR,commande refusee"); return; }
  Serial.printf("PARAM,F=%d,S=%d,R=%d,T=%d\n", frequenceCible, seuil, refractaire, frequenceTest);
}

// ---------- démarrage ----------
void setup() {
  Serial.begin(115200);
  if (esp_reset_reason() == ESP_RST_TASK_WDT) Serial.println("ALERTE,redemarrage par le watchdog");
  pinMode(PIN_STIM, OUTPUT);
  pinMode(PIN_DET, OUTPUT);
  analogReadResolution(12);

  timerEch = timerBegin(1000000);                  // 1 pas = 1 µs
  timerAttachInterrupt(timerEch, &onEch);
  timerAlarm(timerEch, 2000, true, 0);             // toutes les 2 ms
  timerStim = timerBegin(1000000);
  timerAttachInterrupt(timerStim, &onStim);
  reglerFrequence(frequenceCible);

  esp_task_wdt_config_t config = { .timeout_ms = 2000, .idle_core_mask = 0, .trigger_panic = true };
  esp_task_wdt_reconfigure(&config);              // question 38
  esp_task_wdt_add(NULL);
  Serial.println("PRET,mode VVI");
}

// ---------- boucle ----------
void loop() {
  unsigned long t = millis();
  if (tickEch) {                                   // question 32 : un échantillon toutes les 2 ms
    tickEch = false;
    genererECG();
    int v = analogRead(PIN_ECG);
    surveillerECG(v, t);
    if (detecter(v, t)) battementDetecte(t);       // question 33
    esp_task_wdt_reset();                          // nourri seulement si un échantillon a été traité
  }
  if (tickStim) {                                  // intervalle écoulé sans battement
    tickStim = false;
    stimuler(t);
  }
  if (finStim != 0 && t >= finStim) { digitalWrite(PIN_STIM, LOW); finStim = 0; }
  if (finDet != 0 && t >= finDet) { digitalWrite(PIN_DET, LOW); finDet = 0; }
  lireSerie();
}

Avec T0, plus aucune onde R : une stimulation part 1 000 ms après le dernier battement, puis toutes les 1 000 ms (60 bpm). L'intervalle RR de 858 ms vient de l'ECG de test : \(60\,000 / 70 \approx 857\) ms, arrondi au pas de 2 ms. Les instants exacts dépendent du démarrage.

Deux limites à écrire dans le rapport :

  • le compteur phase de l'ECG de test est arrondi à 2 ms : la fréquence de test réelle est à 2 ms près ;
  • la LED dure 20 ms pour être vue ; une vraie impulsion dure moins d'une milliseconde et son amplitude se règle (D/A).

Compiler et téléverser

  • Arduino IDE, paquet esp32 d'Espressif en version 3 (les fonctions timerBegin(fréquence), timerAlarm et esp_task_wdt_reconfigure n'existent pas en version 2) ;
  • carte DOIT ESP32 DEVKIT V1, port de la carte, puis Téléverser ;
  • moniteur série à 115 200 bauds, fin de ligne « Nouvelle ligne » pour les commandes.

À retenir

  • Une seule carte : le DAC fabrique l'ECG, l'ADC le relit, deux timers donnent le temps, deux LED montrent détection et stimulation.
  • Toute la logique VVI : battement détecté, le timer de stimulation repart de 0 ; alarme du timer, on stimule.
  • Les lignes série sont les traces de tests du rapport.
